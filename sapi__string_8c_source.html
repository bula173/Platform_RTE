101{

131 data = (char *)dest->buf.data;
132 if (i > 0U)

134 (void)memcpy(&data[dest->buf.length], src, i);

312static void sapi_string_format_u64_digits(uint64_t value, char *out, size_t *out_len)
313{
314 char rev[20];
315 size_t n = 0U;
316 uint64_t v = value;
317
318 if (v == 0U)
319 {
320 out[0] = '0';
321 *out_len = 1U;
322 return;
323 }
324 while (v > 0U)
325 {
326 char digit = (char)(v % 10U);
327 rev[n] = (char)('0' + digit);
328 n++;
329 v /= 10U;
330 }
331 {
332 size_t i;
333 for (i = 0U; i < n; i++)
334 {
335 out[i] = rev[(n - 1U) - i];
336 }
337 }
338 *out_len = n;
339}

424sapi_status_t sapi_string_to_u64(const sapi_string_t *str, uint64_t *out_value)
425{
426 size_t i;
427 uint64_t acc = 0U;
428 const char *data;
429
430 if ((str == NULL) || (out_value == NULL))
431 {
432 return SAPI_STATUS_INVALID_PARAM;
433 }
434 if (!sapi_buffer_is_valid(&str->buf))
435 {
436 return SAPI_STATUS_INVALID_PARAM;
437 }
438 if (str->buf.length == 0U)
439 {
440 return SAPI_STATUS_INVALID_PARAM;
441 }
442 data = (const char *)str->buf.data;
443 for (i = 0U; i < str->buf.length; i++)
444 {
445 char c = data[i];
446 uint64_t digit;
447
448 if ((c < '0') || (c > '9'))
449 {
450 return SAPI_STATUS_INVALID_PARAM;
451 }
452 digit = (uint64_t)(c - '0');
453 if (acc > ((UINT64_MAX - digit) / 10U))
454 {
455 return SAPI_STATUS_VALUE_OUT_OF_RANGE;
456 }
457 acc = (acc * 10U) + digit;
458 }
459 *out_value = acc;
460 return SAPI_STATUS_OK;
461}

463sapi_status_t sapi_string_to_i64(const sapi_string_t *str, int64_t *out_value)
464{
465 size_t i;
466 size_t start;
467 bool negative;
468 uint64_t acc = 0U;
469 const char *data;
470
471 if ((str == NULL) || (out_value == NULL))
472 {
473 return SAPI_STATUS_INVALID_PARAM;
474 }
475 if (!sapi_buffer_is_valid(&str->buf))
476 {
477 return SAPI_STATUS_INVALID_PARAM;
478 }
479 if (str->buf.length == 0U)
480 {
481 return SAPI_STATUS_INVALID_PARAM;
482 }
483 data = (const char *)str->buf.data;
484 negative = false;
485 start = 0U;
486 if (data[0] == '-')
487 {
488 negative = true;
489 start = 1U;
490 }
491 if (start >= str->buf.length)
492 {
493 return SAPI_STATUS_INVALID_PARAM; /* "-" with no digits */
494 }
495 for (i = start; i < str->buf.length; i++)
496 {
497 char c = data[i];
498 uint64_t digit;
499
500 if ((c < '0') || (c > '9'))
501 {
502 return SAPI_STATUS_INVALID_PARAM;
503 }
504 digit = (uint64_t)(c - '0');
505 if (acc > ((UINT64_MAX - digit) / 10U))
506 {
507 return SAPI_STATUS_VALUE_OUT_OF_RANGE;
508 }
509 acc = (acc * 10U) + digit;
510 }
511 if (negative)
512 {
513 uint64_t min_magnitude = (uint64_t)INT64_MAX + 1U;
514 if (acc > min_magnitude)
515 {
516 return SAPI_STATUS_VALUE_OUT_OF_RANGE;
517 }
518 if (acc == min_magnitude)
519 {
520 *out_value = INT64_MIN;
521 }
522 else
523 {
524 *out_value = -(int64_t)acc;
525 }
526 }
527 else
528 {
529 if (acc > (uint64_t)INT64_MAX)
530 {
531 return SAPI_STATUS_VALUE_OUT_OF_RANGE;
532 }
533 *out_value = (int64_t)acc;
534 }
535 return SAPI_STATUS_OK;
536}
