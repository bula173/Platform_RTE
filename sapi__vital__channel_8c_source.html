48static bool sapi_vital_channel_has_quorum(sapi_vital_channel_t *handle)
49{
50 uint32_t healthy_count = 0;
51
52 if (handle == NULL) {
53 return false;
54 }
55
56 for (uint32_t i = 0; i < handle->channel_count; i++) {
57 if (handle->health[i].is_healthy) {
58 healthy_count++;
59 }
60 }
61
62 switch (handle->config.voting_strategy) {
63 case SAPI_VOTING_2OO2:
64 return healthy_count >= 2;
65 case SAPI_VOTING_2OO3:
66 return healthy_count >= 2;
67 case SAPI_VOTING_NMR:
68 return healthy_count >= handle->config.quorum_size;
69 default:
70 return false;
71 }
72}

74sapi_status_t sapi_vital_channel_init(sapi_vital_channel_t *storage,
75 const sapi_vital_channel_config_t *config,
76 void **channels,
77 uint32_t channel_count)
78{
79 if (storage == NULL || config == NULL || channels == NULL) {
80 return SAPI_STATUS_INVALID_PARAM;
81 }
82
83 /* Validate backend callbacks (transport-agnostic mechanism) */
84 if (config->backend_send == NULL || config->backend_recv == NULL) {
85 return SAPI_STATUS_INVALID_PARAM;
86 }
87
88 if (channel_count < 2 || channel_count > SAPI_VITAL_CHANNEL_MAX_CHANNELS) {
89 return SAPI_STATUS_INVALID_PARAM;
90 }
91
92 /* Validate voting strategy */
93 switch (config->voting_strategy) {
94 case SAPI_VOTING_2OO2:
95 if (channel_count != 2) {
96 return SAPI_STATUS_INVALID_PARAM;
97 }
98 break;
99 case SAPI_VOTING_2OO3:
100 if (channel_count != 3) {
101 return SAPI_STATUS_INVALID_PARAM;
102 }
103 break;
104 case SAPI_VOTING_NMR:
105 if (config->quorum_size > channel_count || config->quorum_size < 2) {
106 return SAPI_STATUS_INVALID_PARAM;
107 }
108 break;
109 default:
110 return SAPI_STATUS_INVALID_PARAM;
111 }
112
113 /* Initialize storage */
114 storage->config = *config;
115 storage->channels = channels;
116 storage->channel_count = channel_count;
117
118 /* Initialize health tracking for all channels */
119 for (uint32_t i = 0; i < channel_count; i++) {
120 storage->health[i].send_count = 0;
121 storage->health[i].send_error_count = 0;
122 storage->health[i].receive_count = 0;
123 storage->health[i].receive_error_count = 0;
124 storage->health[i].disagreement_count = 0;
125 storage->health[i].is_healthy = true;
126 storage->health[i].last_error = SAPI_STATUS_OK;
127 }
128
129 return SAPI_STATUS_OK;
130}

132sapi_status_t sapi_vital_channel_send(sapi_vital_channel_t *handle,
133 const void *data,
134 size_t data_size)
135{
136 sapi_status_t rc;
137 bool all_success = true;
138
139 if (handle == NULL || data == NULL || data_size == 0) {
140 return SAPI_STATUS_INVALID_PARAM;
141 }
142
143 if (data_size > SAPI_VITAL_CHANNEL_MAX_MESSAGE_SIZE) {
144 return SAPI_STATUS_RESOURCE_EXHAUSTED;
145 }
146
147 if (!sapi_vital_channel_has_quorum(handle)) {
148 return SAPI_STATUS_HARDWARE_FAULT;
149 }
150
151 /* Broadcast to all redundant channels (atomic send via backend callback) */
152 for (uint32_t i = 0; i < handle->channel_count; i++) {
153 if (!handle->health[i].is_healthy) {
154 continue;
155 }
156
157 rc = handle->config.backend_send(handle->channels[i], data, data_size);
158 if (rc != SAPI_STATUS_OK) {
159 handle->health[i].send_error_count++;
160 handle->health[i].last_error = rc;
161 all_success = false;
162
163 if (rc == SAPI_STATUS_TIMEOUT) {
164 return SAPI_STATUS_TIMEOUT;
165 }
166 } else {
167 handle->health[i].send_count++;
168 }
169 }
170
171 /* All-or-nothing semantics: all channels must succeed */
172 if (!all_success) {
173 return SAPI_STATUS_HARDWARE_FAULT;
174 }
175
176 return SAPI_STATUS_OK;
177}

179sapi_status_t sapi_vital_channel_receive(sapi_vital_channel_t *handle,
180 void *data,
181 size_t data_size,
182 sapi_voting_result_t *result,
183 size_t *bytes_received)
184{
185 uint32_t successful_receives = 0;
186 sapi_voting_result_t vote_result = SAPI_VOTING_AGREED;
187 sapi_status_t rc;
188 uint8_t voting_buffers[SAPI_VITAL_CHANNEL_MAX_CHANNELS][SAPI_VITAL_CHANNEL_MAX_MESSAGE_SIZE];
189 uint32_t first_agree_idx = 0;
190 bool first_data_valid = false;
191
192 if (handle == NULL || data == NULL || data_size == 0) {
193 return SAPI_STATUS_INVALID_PARAM;
194 }
195
196 if (data_size > SAPI_VITAL_CHANNEL_MAX_MESSAGE_SIZE) {
197 return SAPI_STATUS_RESOURCE_EXHAUSTED;
198 }
199
200 if (!sapi_vital_channel_has_quorum(handle)) {
201 if (result != NULL) {
202 *result = SAPI_VOTING_INSUFFICIENT_QUORUM;
203 }
204 if (bytes_received != NULL) {
205 *bytes_received = 0;
206 }
207 return SAPI_STATUS_HARDWARE_FAULT;
208 }
209
210 /* Receive from all redundant channels and perform voting */
211 for (uint32_t i = 0; i < handle->channel_count; i++) {
212 if (!handle->health[i].is_healthy) {
213 continue;
214 }
215
216 rc = handle->config.backend_recv(handle->channels[i], voting_buffers[i],
217 data_size, handle->config.channel_timeout_ms);
218
219 if (rc == SAPI_STATUS_OK) {
220 handle->health[i].receive_count++;
221 successful_receives++;
222
223 /* Store first successful receive for comparison */
224 if (!first_data_valid) {
225 first_data_valid = true;
226 first_agree_idx = i;
227 } else {
228 /* Compare with first channel's data (voting) */
229 if (!sapi_vital_channel_data_equal(voting_buffers[i], voting_buffers[first_agree_idx],
230 data_size)) {
231 handle->health[i].disagreement_count++;
232 handle->health[first_agree_idx].disagreement_count++;
233 vote_result = SAPI_VOTING_DISAGREED;
234 }
235 }
236 } else {
237 handle->health[i].receive_error_count++;
238 handle->health[i].last_error = rc;
239
240 if (rc == SAPI_STATUS_TIMEOUT) {
241 vote_result = SAPI_VOTING_TIMEOUT;
242 }
243 }
244 }
245
246 /* Check if we have enough successful receives for voting */
247 if (successful_receives < 2) {
248 vote_result = (successful_receives == 0) ? SAPI_VOTING_TIMEOUT
249 : SAPI_VOTING_INSUFFICIENT_QUORUM;
250
251 if (result != NULL) {
252 *result = vote_result;
253 }
254
255 if (bytes_received != NULL) {
256 *bytes_received = 0;
257 }
258
259 return SAPI_STATUS_HARDWARE_FAULT;
260 }
261
262 /* Copy agreed data to output buffer */
263 if (vote_result == SAPI_VOTING_AGREED && first_data_valid) {
264 memcpy(data, voting_buffers[first_agree_idx], data_size);
265 if (bytes_received != NULL) {
266 *bytes_received = data_size;
267 }
268 } else {
269 /* Disagreement or no data available */
270 if (handle->config.log_disagreements && vote_result == SAPI_VOTING_DISAGREED) {
271 sapi_log_write(SAPI_LOG_LEVEL_ERROR, "VITAL_CHANNEL",
272 "Redundant channel voting disagreement detected");
273 }
274
275 /* Trigger safe-state on disagreement */
276 if (vote_result == SAPI_VOTING_DISAGREED) {
277 SAPI_SAFESTATE(SAPI_SAFESTATE_LEVEL_SAFE,
278 SAPI_SAFESTATE_REASON_UNSPECIFIED);
279 }
280
281 if (bytes_received != NULL) {
282 *bytes_received = 0;
283 }
284 }
285
286 /* Invoke disagreement callback if configured */
287 if (vote_result != SAPI_VOTING_AGREED && handle->config.on_disagreement != NULL) {
288 handle->config.on_disagreement(handle->config.context, &vote_result);
289 }
290
291 if (result != NULL) {
292 *result = vote_result;
293 }
294
295 return (vote_result == SAPI_VOTING_AGREED) ? SAPI_STATUS_OK
296 : SAPI_STATUS_HARDWARE_FAULT;
297}
