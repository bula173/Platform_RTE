74static sapi_status_t dual_channel_poll_link_once(sapi_dual_channel_t *channel, uint32_t link_index,
75 sapi_duration_ms_t timeout_ms, dual_poll_result_t *out_result)
76{
77 uint8_t raw[SAPI_DUAL_MSGCHANNEL_MAX_PAYLOAD];
78 uint8_t raw_size = 0U;
79 uint32_t sequence = 0U;
80 sapi_status_t status;
81 sapi_dual_frame_header_t header;
82
83 out_result->matched = false;
84
85 status = sapi_dual_msgchannel_receive(&channel->links[link_index], raw, (uint8_t)sizeof(raw), timeout_ms,
86 &raw_size, &sequence);
87 if (status != SAPI_STATUS_OK)
88 {
89 return status;
90 }
91 if (raw_size < (uint8_t)sizeof(header))
92 {
93 /* Too short to even hold this layer's own header - treat as a
94 * defended-integrity failure, same family as a Layer-1 CRC/
95 * sequence failure, not a silently-ignored malformed frame. */
96 return SAPI_STATUS_DATA_CORRUPTION;
97 }
98 (void)memcpy(&header, raw, sizeof(header));
99
100 switch ((sapi_dual_frame_kind_t)header.kind)
101 {
102 case SAPI_DUAL_FRAME_KIND_DATA:
103 {
104 uint8_t app_size = (uint8_t)(raw_size - (uint8_t)sizeof(header));
105 sapi_dual_ack_frame_t ack;
106
107 if (app_size <= SAPI_DUAL_CHANNEL_MAX_PAYLOAD)
108 {
109 (void)memcpy(channel->pending_data, &raw[sizeof(header)], app_size);
110 channel->pending_data_size = app_size;
111 channel->pending_data_valid = true;
112 }
113
114 /* Auto-ACK, best-effort/fire-and-forget: a lost ACK is
115 * observable to the sender as its own timeout on this link
116 * (ADR-020 section 2), not silently hidden here. */
117 ack.header.kind = (uint8_t)SAPI_DUAL_FRAME_KIND_ACK;
118 ack.header.reserved[0] = 0U;
119 ack.header.reserved[1] = 0U;
120 ack.header.reserved[2] = 0U;
121 ack.acked_sequence = sequence;
122 (void)sapi_dual_msgchannel_send(&channel->links[link_index], (const uint8_t *)&ack, (uint8_t)sizeof(ack),
123 channel->ack_timeout_ms, NULL);
124
125 out_result->matched = true;
126 out_result->kind = SAPI_DUAL_FRAME_KIND_DATA;
127 break;
128 }
129 case SAPI_DUAL_FRAME_KIND_ACK:
130 {
131 sapi_dual_ack_frame_t ack;
132
133 if (raw_size < (uint8_t)sizeof(ack))
134 {
135 return SAPI_STATUS_DATA_CORRUPTION;
136 }
137 (void)memcpy(&ack, raw, sizeof(ack));
138
139 out_result->matched = true;
140 out_result->kind = SAPI_DUAL_FRAME_KIND_ACK;
141 out_result->ack_sequence = ack.acked_sequence;
142 break;
143 }
144 case SAPI_DUAL_FRAME_KIND_STATE:
145 {
146 if (raw_size < (uint8_t)sizeof(sapi_dual_state_frame_t))
147 {
148 return SAPI_STATUS_DATA_CORRUPTION;
149 }
150 (void)memcpy(&channel->pending_state, raw, sizeof(channel->pending_state));
151 channel->pending_state_valid = true;
152
153 out_result->matched = true;
154 out_result->kind = SAPI_DUAL_FRAME_KIND_STATE;
155 break;
156 }
157 default:
158 /* Unrecognized kind - defensive only (not reachable through
159 * a conforming sender), ignore rather than fail. */
160 break;
161 }
162
163 return SAPI_STATUS_OK;
164}

215sapi_status_t sapi_dual_channel_send(sapi_dual_channel_t *channel, const uint8_t *payload, uint8_t payload_size,
216 uint32_t *out_ack_link_count)
217{
218 uint8_t frame[(size_t)SAPI_DUAL_CHANNEL_MAX_PAYLOAD + sizeof(sapi_dual_frame_header_t)];
219 uint8_t frame_size;
220 uint32_t i;
221 uint32_t ack_count = 0U;
222 /* REQ-DUAL-CHANNEL-008: a link whose own send/receive reports
223 * something other than SAPI_STATUS_OK/SAPI_STATUS_TIMEOUT (e.g.
224 * SAPI_STATUS_HARDWARE_FAULT from a closed/reset connection) is
225 * genuinely broken, not just quiet - see this function's own header
226 * for why collapsing that into the same generic TIMEOUT every other
227 * "no ACK yet" case returns hid real transport failures from every
228 * caller. Kept as the FIRST such status seen across all links this
229 * call, not the last - an arbitrary but stable choice among possibly
230 * several simultaneous failures. */
231 bool saw_hard_fault = false;
232 sapi_status_t hard_fault_status = SAPI_STATUS_OK;
233
234 if (channel == NULL)
235 {
236 return SAPI_STATUS_INVALID_PARAM;
237 }
238 if ((payload == NULL) && (payload_size != 0U))
239 {
240 return SAPI_STATUS_INVALID_PARAM;
241 }
242 if (payload_size > SAPI_DUAL_CHANNEL_MAX_PAYLOAD)
243 {
244 return SAPI_STATUS_INVALID_PARAM;
245 }
246
247 {
248 sapi_dual_frame_header_t header;
249
250 header.kind = (uint8_t)SAPI_DUAL_FRAME_KIND_DATA;
251 header.reserved[0] = 0U;
252 header.reserved[1] = 0U;
253 header.reserved[2] = 0U;
254 (void)memcpy(frame, &header, sizeof(header));
255 if (payload_size > 0U)
256 {
257 (void)memcpy(&frame[sizeof(header)], payload, payload_size);
258 }
259 frame_size = (uint8_t)(sizeof(header) + payload_size);
260 }
261
262 for (i = 0U; i < channel->link_count; i++)
263 {
264 uint32_t sent_sequence = 0U;
265 sapi_status_t send_status;
266 bool link_now_up = false;
267
268 send_status = sapi_dual_msgchannel_send(&channel->links[i], frame, frame_size, channel->ack_timeout_ms,
269 &sent_sequence);
270 if ((send_status == SAPI_STATUS_HARDWARE_FAULT) && (!saw_hard_fault))
271 {
272 saw_hard_fault = true;
273 hard_fault_status = send_status;
274 }
275 if (send_status == SAPI_STATUS_OK)
276 {
277 sapi_duration_ms_t remaining = channel->ack_timeout_ms;
278 sapi_timestamp_ms_t start_ms = 0U;
279 /* Bounds how many consecutive polls may complete without
280 * sapi_timer_now() showing any measurable progress since
281 * start_ms, before this loop gives up on this link for this
282 * round - see the "else" branch below for why this can
283 * legitimately happen more than once and must not itself be
284 * unbounded. SAPI_DUAL_CHANNEL_STALL_POLL_LIMIT is a fixed,
285 * generous cap (a real exchange needs at most a handful of
286 * iterations - one per DATA/STATE/foreign-ACK frame handled
287 * as a side effect before this link's own matching ACK
288 * arrives), not a tuned timing value. */
289 uint32_t stall_polls = 0U;
290
291 (void)sapi_timer_now(&start_ms);
292
293 while ((remaining > 0U) && (stall_polls < SAPI_DUAL_CHANNEL_STALL_POLL_LIMIT))
294 {
295 dual_poll_result_t result;
296 sapi_status_t poll_status;
297 sapi_timestamp_ms_t now_ms = 0U;
298
299 poll_status = dual_channel_poll_link_once(channel, i, remaining, &result);
300 if ((poll_status == SAPI_STATUS_OK) && result.matched && (result.kind == SAPI_DUAL_FRAME_KIND_ACK)
301 && (result.ack_sequence == sent_sequence))
302 {
303 link_now_up = true;
304 break;
305 }
306 if (poll_status == SAPI_STATUS_HARDWARE_FAULT)
307 {
308 /* Genuinely broken (not just "no ACK frame arrived
309 * yet this poll"), e.g. HARDWARE_FAULT from the peer
310 * having closed/reset the connection - see this
311 * function's own header. No point continuing to poll
312 * a dead link for the rest of its own ack_timeout_ms
313 * budget. */
314 if (!saw_hard_fault)
315 {
316 saw_hard_fault = true;
317 hard_fault_status = poll_status;
318 }
319 break;
320 }
321
322 /* Recompute the remaining budget from wall-clock elapsed
323 * time, not a fixed per-iteration decrement - a DATA/STATE
324 * frame handled as a side effect above may have consumed
325 * an arbitrary fraction of this link's own timeout
326 * already. */
327 (void)sapi_timer_now(&now_ms);
328 if (now_ms > start_ms)
329 {
330 sapi_timestamp_ms_t elapsed = now_ms - start_ms;
331
332 if (elapsed >= (sapi_timestamp_ms_t)channel->ack_timeout_ms)
333 {
334 remaining = 0U;
335 }
336 else
337 {
338 remaining = channel->ack_timeout_ms - (sapi_duration_ms_t)elapsed;
339 }
340 stall_polls = 0U;
341 }
342 else
343 {
344 /* REQ-DUAL-CHANNEL-007: now_ms == start_ms - no
345 * measurable time has passed on sapi_timer_now()'s own
346 * tick resolution since this round started. This used
347 * to be treated as "no timer
348 * backend available, give up after one attempt"
349 * (REQ-OAL-LOG-001-style "never spin on an
350 * unmeasurable interval"), but a real localhost round
351 * trip (connect, send DATA, receive the peer's own
352 * DATA, auto-ACK it, receive the peer's own ACK)
353 * routinely completes inside a single millisecond
354 * tick, which made that same guard misfire as a false
355 * "no timer" abort after exactly one poll - discovered
356 * live via SITE's migration to sapi_safechannel
357 * (ADR-022), where both sites send their negotiation
358 * beacon at nearly the same instant over a real
359 * loopback TCP link. `remaining` is left unchanged so
360 * a fast exchange like that one gets the extra polls
361 * it needs; stall_polls bounds how many such
362 * no-progress iterations are allowed before this loop
363 * gives up anyway, so a link with a genuinely
364 * non-advancing or absent timer still cannot spin
365 * forever. */
366 stall_polls++;
367 }
368 }
369 }
370
371 channel->link_up[i] = link_now_up;
372 if (link_now_up)
373 {
374 ack_count++;
375 }
376 }
377
378 if (out_ack_link_count != NULL)
379 {
380 *out_ack_link_count = ack_count;
381 }
382
383 dual_channel_update_status(channel);
384
385 if (ack_count > 0U)
386 {
387 return SAPI_STATUS_OK;
388 }
389 return saw_hard_fault ? hard_fault_status : SAPI_STATUS_TIMEOUT;
390}

392sapi_status_t sapi_dual_channel_receive(sapi_dual_channel_t *channel, uint8_t *out_payload, uint8_t max_size,
393 sapi_duration_ms_t timeout_ms, uint8_t *out_size)
394{
395 if ((channel == NULL) || (out_payload == NULL) || (out_size == NULL) || (max_size == 0U))
396 {
397 return SAPI_STATUS_INVALID_PARAM;
398 }
399
400 {
401 bool saw_hard_fault = false;
402 sapi_status_t hard_fault_status = SAPI_STATUS_OK;
403
404 if (!channel->pending_data_valid)
405 {
406 uint32_t i;
407 sapi_duration_ms_t per_link_timeout = (channel->link_count > 0U) ? (timeout_ms / channel->link_count) : 0U;
408
409 /* Sweeps every configured link every call, even after an
410 * earlier link in this same sweep already staged a DATA frame -
411 * stopping early here would let a link with a shorter path (or
412 * one that always has traffic) starve every other redundant
413 * link of its own auto-ACK indefinitely (ADR-020 section 2: a
414 * redundant link should degrade to DOWN only from genuinely not
415 * responding, never from this module never getting around to
416 * polling it). Still swept to completion even after a hard
417 * fault on an earlier link, for the same reason. */
418 for (i = 0U; i < channel->link_count; i++)
419 {
420 dual_poll_result_t result;
421 sapi_status_t poll_status;
422
423 poll_status = dual_channel_poll_link_once(channel, i, per_link_timeout, &result);
424 /* REQ-DUAL-CHANNEL-008: see sapi_dual_channel_send()'s own
425 * doc on this same distinction - a link reporting
426 * something other than OK/TIMEOUT (e.g. HARDWARE_FAULT
427 * from a closed/reset connection) is broken, not just
428 * quiet, and that must not be silently collapsed into the
429 * same generic TIMEOUT "nothing staged yet" returns below. */
430 if ((poll_status == SAPI_STATUS_HARDWARE_FAULT) && (!saw_hard_fault))
431 {
432 saw_hard_fault = true;
433 hard_fault_status = poll_status;
434 }
435 }
436 }
437
438 if (!channel->pending_data_valid)
439 {
440 return saw_hard_fault ? hard_fault_status : SAPI_STATUS_TIMEOUT;
441 }
442 }
443 if (channel->pending_data_size > max_size)
444 {
445 /* Caller's buffer is too small for a staged frame - reported,
446 * not silently truncated (safety-relevant data). */
447 return SAPI_STATUS_INVALID_PARAM;
448 }
449
450 (void)memcpy(out_payload, channel->pending_data, channel->pending_data_size);
451 *out_size = channel->pending_data_size;
452 channel->pending_data_valid = false;
453
454 return SAPI_STATUS_OK;
455}
