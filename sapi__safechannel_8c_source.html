34static sapi_status_t safechannel_open_one_endpoint(sapi_netlink_storage_t *storage,
35 const sapi_safechannel_endpoint_t *endpoint, size_t message_size,
36 sapi_duration_ms_t connect_timeout_ms,
37 sapi_netlink_handle_t *out_handle)
38{
39 sapi_netlink_config_t cfg;
40 sapi_status_t status;
41
42 memset(&cfg, 0, sizeof(cfg));
43 cfg.role = endpoint->role;
44 cfg.host = endpoint->host;
45 cfg.port = endpoint->port;
46 cfg.message_size = message_size;
47 cfg.connect_timeout_ms = connect_timeout_ms;
48
49 /* sapi_netlink_open() itself already bounds a single attempt to
50 * connect_timeout_ms (REQ-OAL-NETLINK-002); a LISTEN-role endpoint
51 * blocks until accepted or that same timeout, so one attempt is
52 * sufficient there. A CONNECT-role endpoint can fail fast if the
53 * peer isn't listening yet, so this retries for the same overall
54 * budget rather than failing on the first attempt. */
55 status = sapi_netlink_open(storage, &cfg, out_handle);
56 return status;
57}

208sapi_status_t sapi_safechannel_open(sapi_safechannel_t *channel, const sapi_safechannel_config_t *config)
209{
210 sapi_status_t status;
211
212 if ((channel == NULL) || (config == NULL))
213 {
214 return SAPI_STATUS_INVALID_PARAM;
215 }
216 memset(channel, 0, sizeof(*channel));
217 channel->type = config->type;
218
219 switch (config->type)
220 {
221 case SAPI_SAFECHANNEL_TYPE_DUAL_REDUNDANT:
222 status = safechannel_open_dual(channel, &config->as.dual);
223 break;
224 case SAPI_SAFECHANNEL_TYPE_VITAL_VOTED:
225 status = safechannel_open_vital(channel, &config->as.vital);
226 break;
227 default:
228 status = SAPI_STATUS_INVALID_PARAM;
229 break;
230 }
231
232 channel->is_open = (status == SAPI_STATUS_OK);
233 return status;
234}

283sapi_safechannel_link_status_t sapi_safechannel_get_status(const sapi_safechannel_t *channel)
284{
285 if ((channel == NULL) || (!channel->is_open))
286 {
287 return SAPI_SAFECHANNEL_LINK_DOWN;
288 }
289
290 if (channel->type == SAPI_SAFECHANNEL_TYPE_DUAL_REDUNDANT)
291 {
292 switch (sapi_dual_channel_get_status(&channel->impl.dual))
293 {
294 case SAPI_DUAL_CHANNEL_STATUS_FULL:
295 return SAPI_SAFECHANNEL_LINK_FULL;
296 case SAPI_DUAL_CHANNEL_STATUS_DEGRADED:
297 return SAPI_SAFECHANNEL_LINK_DEGRADED;
298 case SAPI_DUAL_CHANNEL_STATUS_DOWN:
299 default:
300 return SAPI_SAFECHANNEL_LINK_DOWN;
301 }
302 }
303 {
304 uint32_t healthy_count = 0U;
305
306 (void)sapi_voter_get_aggregated_health(&channel->impl.vital.voter,
307 &healthy_count, NULL);
308 if (healthy_count == 0U)
309 {
310 return SAPI_SAFECHANNEL_LINK_DOWN;
311 }
312 if (healthy_count < channel->link_count)
313 {
314 return SAPI_SAFECHANNEL_LINK_DEGRADED;
315 }
316 return SAPI_SAFECHANNEL_LINK_FULL;
317 }

320sapi_status_t sapi_safechannel_close(sapi_safechannel_t *channel)
321{
322 if (channel == NULL)
323 {
324 return SAPI_STATUS_OK;
325 }
326 if (!channel->is_open)
327 {
328 return SAPI_STATUS_OK;
329 }
330
331 if (channel->type == SAPI_SAFECHANNEL_TYPE_VITAL_VOTED)
332 {
333 uint32_t i;
334
335 for (i = 0U; i < channel->link_count; i++)
336 {
337 (void)sapi_channel_destroy(&channel->impl.vital.channels[i]);
338 }
339 (void)sapi_voter_destroy(&channel->impl.vital.voter);
340 }
341 safechannel_close_links(channel, channel->link_count);
342 channel->link_count = 0U;
343 channel->is_open = false;
344 return SAPI_STATUS_OK;
345}
