49int sapi_appmanager_run(const sapi_appmanager_config_t *config)
50{
51 sapi_status_t status;
52
53 /* Validate configuration */
54 if (config == NULL || config->ops == NULL) {
55 fprintf(stderr, "ERROR: Invalid sapi_appmanager_config_t\n");
56 return EXIT_FAILURE;
57 }
58
59 if (config->ops->init == NULL ||
60 config->ops->execute == NULL ||
61 config->ops->shutdown == NULL ||
62 config->ops->get_name == NULL ||
63 config->ops->get_version == NULL) {
64 fprintf(stderr, "ERROR: Incomplete sapi_appmanager_operations_t\n");
65 return EXIT_FAILURE;
66 }
67
68 /* Initialize application manager state */
69 g_app_state.state = SAPI_APP_STATE_INITIALIZING;
70 g_app_state.iteration_count = 0;
71 g_app_state.error_count = 0;
72 g_app_state.last_error = SAPI_STATUS_OK;
73 g_shutdown_requested = 0;
74
75 /* Print banner */
76 printf("\n");
77 printf("═══════════════════════════════════════════════════════════════\n");
78 printf("%s v%s\n", config->ops->get_name(), config->ops->get_version());
79 printf("═══════════════════════════════════════════════════════════════\n");
80 printf("\n");
81
82 /* ========================================================================
83 * INITIALIZATION PHASE
84 * ======================================================================== */
85
86 status = config->ops->init(config->context);
87
88 if (status != SAPI_STATUS_OK) {
89 fprintf(stderr, "[APPMANAGER ERROR] Initialization failed: %d\n", status);
90 g_app_state.state = SAPI_APP_STATE_ERROR;
91 g_app_state.last_error = status;
92 g_app_state.error_count++;
93
94 /* Always attempt shutdown even on init failure */
95 config->ops->shutdown(config->context);
96
97 return EXIT_FAILURE;
98 }
99
100 g_app_state.state = SAPI_APP_STATE_RUNNING;
101
102 /* ========================================================================
103 * EXECUTION PHASE
104 * ======================================================================== */
105
106 while (!g_shutdown_requested &&
107 (config->max_iterations == 0 ||
108 g_app_state.iteration_count < config->max_iterations)) {
109
110 g_app_state.iteration_count++;
111
112 /* Execute one iteration of application work */
113 status = config->ops->execute(config->context);
114
115 /* Handle errors */
116 if (status != SAPI_STATUS_OK) {
117 g_app_state.error_count++;
118 g_app_state.last_error = status;
119
120 /* Check error threshold */
121 if (config->error_threshold > 0 &&
122 g_app_state.error_count >= config->error_threshold) {
123 fprintf(stderr,
124 "[APPMANAGER ERROR] Error threshold exceeded (%u/%u), shutting down\n",
125 g_app_state.error_count,
126 config->error_threshold);
127 g_shutdown_requested = 1;
128 }
129 }
130 }
131
132 /* ========================================================================
133 * SHUTDOWN PHASE
134 * ======================================================================== */
135
136 g_app_state.state = SAPI_APP_STATE_SHUTTING_DOWN;
137
138 status = config->ops->shutdown(config->context);
139
140 if (status != SAPI_STATUS_OK) {
141 fprintf(stderr, "[APPMANAGER WARN] Shutdown returned non-OK status: %d\n", status);
142 /* Continue anyway; shutdown must complete */
143 }
144
145 g_app_state.state = SAPI_APP_STATE_SHUTDOWN;
146
147 /* Print final summary */
148 printf("\n");
149 printf("═══════════════════════════════════════════════════════════════\n");
150 printf("%s Summary\n", config->ops->get_name());
151 printf("═══════════════════════════════════════════════════════════════\n");
152 printf("Iterations executed: %u\n", g_app_state.iteration_count);
153 printf("Errors encountered: %u\n", g_app_state.error_count);
154 printf("Final state: %s\n",
155 (g_app_state.error_count > 0 ? "ERROR" : "OK"));
156 printf("═══════════════════════════════════════════════════════════════\n");
157 printf("\n");
158
159 /* Determine exit code */
160 if (g_app_state.error_count > 0 && config->error_threshold > 0) {
161 return EXIT_FAILURE;
162 }
163
164 return EXIT_SUCCESS;
165}

189/* Async-signal-safe: writes one volatile int, nothing else (no I/O, no
190 * allocation) - see sapi_appmanager_request_shutdown()'s own definition
191 * above. */
192static void sapi_appmanager_signal_handler(int signum)
193{
194 (void)signum;
195 sapi_appmanager_request_shutdown();
196}

198sapi_status_t sapi_appmanager_install_default_signal_handlers(void)
199{
200 struct sigaction sa;
201
202 memset(&sa, 0, sizeof(sa));
203 sa.sa_handler = sapi_appmanager_signal_handler;
204 if (sigemptyset(&sa.sa_mask) != 0)
205 {
206 return SAPI_STATUS_INTERNAL_ERROR;
207 }
208 sa.sa_flags = 0;
209
210 if (sigaction(SIGINT, &sa, NULL) != 0)
211 {
212 return SAPI_STATUS_INTERNAL_ERROR;
213 }
214 if (sigaction(SIGTERM, &sa, NULL) != 0)
215 {
216 return SAPI_STATUS_INTERNAL_ERROR;
217 }
218 return SAPI_STATUS_OK;
219}
