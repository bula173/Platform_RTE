62int sapi_appmanager_run(const sapi_appmanager_config_t *config)
63{
64 sapi_status_t status;
65
66 /* Validate configuration */
67 if (config == NULL || config->ops == NULL) {
68 fprintf(stderr, "ERROR: Invalid sapi_appmanager_config_t\n");
69 return EXIT_FAILURE;
70 }
71
72 if (config->ops->init == NULL ||
73 config->ops->execute == NULL ||
74 config->ops->shutdown == NULL ||
75 config->ops->get_name == NULL ||
76 config->ops->get_version == NULL) {
77 fprintf(stderr, "ERROR: Incomplete sapi_appmanager_operations_t\n");
78 return EXIT_FAILURE;
79 }
80
81 /* Initialize application manager state */
82 g_app_state.state = SAPI_APP_STATE_INITIALIZING;
83 g_app_state.iteration_count = 0;
84 g_app_state.error_count = 0;
85 g_app_state.last_error = SAPI_STATUS_OK;
86 g_shutdown_requested = 0;
87
88 /* Print banner */
89 printf("\n");
90 printf("═══════════════════════════════════════════════════════════════\n");
91 printf("%s v%s\n", config->ops->get_name(), config->ops->get_version());
92 printf("═══════════════════════════════════════════════════════════════\n");
93 printf("\n");
94
95 /* ========================================================================
96 * INITIALIZATION PHASE
97 * ======================================================================== */
98
99 status = config->ops->init(config->context);
100
101 if (status != SAPI_STATUS_OK) {
102 fprintf(stderr, "[APPMANAGER ERROR] Initialization failed: %d\n", status);
103 g_app_state.state = SAPI_APP_STATE_ERROR;
104 g_app_state.last_error = status;
105 g_app_state.error_count++;
106
107 /* Always attempt shutdown even on init failure */
108 config->ops->shutdown(config->context);
109
110 return EXIT_FAILURE;
111 }
112
113 g_app_state.state = SAPI_APP_STATE_RUNNING;
114
115 /* ========================================================================
116 * EXECUTION PHASE
117 * ======================================================================== */
118
119 while (!g_shutdown_requested &&
120 (config->max_iterations == 0 ||
121 g_app_state.iteration_count < config->max_iterations)) {
122
123 g_app_state.iteration_count++;
124
125 /* Execute one iteration of application work */
126 status = config->ops->execute(config->context);
127
128 /* Handle errors */
129 if (status != SAPI_STATUS_OK) {
130 g_app_state.error_count++;
131 g_app_state.last_error = status;
132
133 /* Check error threshold */
134 if (config->error_threshold > 0 &&
135 g_app_state.error_count >= config->error_threshold) {
136 fprintf(stderr,
137 "[APPMANAGER ERROR] Error threshold exceeded (%u/%u), shutting down\n",
138 g_app_state.error_count,
139 config->error_threshold);
140 g_shutdown_requested = 1;
141 }
142 }
143 }
144
145 /* ========================================================================
146 * SHUTDOWN PHASE
147 * ======================================================================== */
148
149 g_app_state.state = SAPI_APP_STATE_SHUTTING_DOWN;
150
151 status = config->ops->shutdown(config->context);
152
153 if (status != SAPI_STATUS_OK) {
154 fprintf(stderr, "[APPMANAGER WARN] Shutdown returned non-OK status: %d\n", status);
155 /* Continue anyway; shutdown must complete */
156 }
157
158 g_app_state.state = SAPI_APP_STATE_SHUTDOWN;
159
160 /* Print final summary */
161 printf("\n");
162 printf("═══════════════════════════════════════════════════════════════\n");
163 printf("%s Summary\n", config->ops->get_name());
164 printf("═══════════════════════════════════════════════════════════════\n");
165 printf("Iterations executed: %u\n", g_app_state.iteration_count);
166 printf("Errors encountered: %u\n", g_app_state.error_count);
167 printf("Final state: %s\n",
168 (g_app_state.error_count > 0 ? "ERROR" : "OK"));
169 printf("═══════════════════════════════════════════════════════════════\n");
170 printf("\n");
171
172 /* Determine exit code */
173 if (g_app_state.error_count > 0 && config->error_threshold > 0) {
174 return EXIT_FAILURE;
175 }
176
177 return EXIT_SUCCESS;
178}

202/* Async-signal-safe: writes one volatile int, nothing else (no I/O, no
203 * allocation) - see sapi_appmanager_request_shutdown()'s own definition
204 * above. */
205static void sapi_appmanager_signal_handler(int signum)
206{
207 (void)signum;
208 sapi_appmanager_request_shutdown();
209}

211sapi_status_t sapi_appmanager_install_default_signal_handlers(void)
212{
213 struct sigaction sa;
214
215 memset(&sa, 0, sizeof(sa));
216 sa.sa_handler = sapi_appmanager_signal_handler;
217 if (sigemptyset(&sa.sa_mask) != 0)
218 {
219 return SAPI_STATUS_INTERNAL_ERROR;
220 }
221 sa.sa_flags = 0;
222
223 if (sigaction(SIGINT, &sa, NULL) != 0)
224 {
225 return SAPI_STATUS_INTERNAL_ERROR;
226 }
227 if (sigaction(SIGTERM, &sa, NULL) != 0)
228 {
229 return SAPI_STATUS_INTERNAL_ERROR;
230 }
231 return SAPI_STATUS_OK;
232}
