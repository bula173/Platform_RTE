75static bool reply_confirms_checkpoint(const sapi_vital_message_t *reply, uint32_t checkpoint_id)
76{
77 uint8_t payload_out[4];
78 uint8_t payload_size_out = 0U;
79 bool confirmed = false;
80 sapi_status_t verify_status = sapi_checksum_vital_message_verify(
81 reply, checkpoint_id, payload_out, sizeof(payload_out), &payload_size_out);
82
83 if ((verify_status == SAPI_STATUS_OK) && (payload_size_out == sizeof(payload_out)))
84 {
85 sapi_buffer_t payload_buf;
86
87 if (sapi_buffer_init(&payload_buf, payload_out, sizeof(payload_out)) == SAPI_STATUS_OK)
88 {
89 uint32_t decoded_id = 0U;
90
91 /* Reads happen against buf->length, not just capacity; the
92 * buffer was just fully written by verify() above, so its
93 * length must be advanced to cover it before reading back. */
94 (void)sapi_buffer_set_length(&payload_buf, sizeof(payload_out));
95
96 if ((sapi_buffer_read_u32_le(&payload_buf, 0U, &decoded_id) == SAPI_STATUS_OK)
97 && (decoded_id == checkpoint_id))
98 {
99 confirmed = true;
100 }
101 }
102 }
103
104 return confirmed;
105}

138sapi_status_t sapi_channel_checkpoint(sapi_vital_channel_t *handle, const sapi_checkpoint_config_t *config)
139{
140 sapi_status_t status = SAPI_STATUS_OK;
141 sapi_vital_message_t arrival_msg;
142
143 if ((handle == NULL) || (config == NULL))
144 {
145 status = SAPI_STATUS_INVALID_PARAM;
146 }
147 else if ((config->expected_node_count == 0U) || (config->expected_node_count > handle->channel_count))
148 {
149 status = SAPI_STATUS_INVALID_PARAM;
150 }
151 else if ((handle->config.backend_send == NULL) || (handle->config.backend_recv == NULL))
152 {
153 /* Defensive: sapi_vital_channel_init() already requires both
154 * callbacks to be non-NULL, but handle is caller-supplied state -
155 * don't trust it blindly just because it was probably initialized
156 * correctly (SIL4 defensive-programming baseline). */
157 status = SAPI_STATUS_NOT_INITIALIZED;
158 }
159 else
160 {
161 status = build_arrival_message(config->checkpoint_id, &arrival_msg);
162 }
163
164 if (status == SAPI_STATUS_OK)
165 {
166 sapi_timestamp_ms_t start_ms = 0U;
167 uint32_t confirmed_count = 0U;
168 uint32_t i;
169
170 (void)sapi_timer_now(&start_ms);
171
172 for (i = 0U; i < handle->channel_count; i++)
173 {
174 sapi_status_t send_status =
175 handle->config.backend_send(handle->channels[i], &arrival_msg, sizeof(arrival_msg));
176
177 if (send_status == SAPI_STATUS_OK)
178 {
179 sapi_vital_message_t reply;
180 sapi_timestamp_ms_t now_ms = start_ms;
181 sapi_status_t recv_status;
182
183 (void)sapi_timer_now(&now_ms);
184 recv_status = handle->config.backend_recv(
185 handle->channels[i], &reply, sizeof(reply),
186 remaining_budget_ms(start_ms, now_ms, config->max_delay_ms));
187
188 if ((recv_status == SAPI_STATUS_OK) && reply_confirms_checkpoint(&reply, config->checkpoint_id))
189 {
190 confirmed_count++;
191 }
192 }
193 }
194
195 if (confirmed_count >= config->expected_node_count)
196 {
197 if (config->watchdog != NULL)
198 {
199 (void)sapi_watchdog_kick(config->watchdog);
200 }
201 status = SAPI_STATUS_OK;
202 }
203 else
204 {
205 /* Matches the pattern sapi_vital_channel_receive() already
206 * uses on a voting disagreement: safe-state is triggered
207 * directly by the sync/vote logic itself, not left to the
208 * caller to notice and react to (REQ-CHECKPOINT-003). */
209 sapi_safestate_enter(SAPI_SAFESTATE_LEVEL_SAFE, SAPI_SAFESTATE_REASON_CHECKPOINT_TIMEOUT, __FILE__,
210 (int32_t)__LINE__, NULL);
211 status = SAPI_STATUS_TIMEOUT;
212 }
213 }
214
215 return status;
216}
