76static bool reply_confirms_checkpoint(const sapi_vital_message_t *reply, uint32_t checkpoint_id)
77{
78 uint8_t payload_out[4];
79 uint8_t payload_size_out = 0U;
80 bool confirmed = false;
81 sapi_status_t verify_status = sapi_checksum_vital_message_verify(
82 reply, checkpoint_id, payload_out, sizeof(payload_out), &payload_size_out);
83
84 if ((verify_status == SAPI_STATUS_OK) && (payload_size_out == sizeof(payload_out)))
85 {
86 sapi_buffer_t payload_buf;
87
88 if (sapi_buffer_init(&payload_buf, payload_out, sizeof(payload_out)) == SAPI_STATUS_OK)
89 {
90 uint32_t decoded_id = 0U;
91
92 /* Reads happen against buf->length, not just capacity; the
93 * buffer was just fully written by verify() above, so its
94 * length must be advanced to cover it before reading back. */
95 (void)sapi_buffer_set_length(&payload_buf, sizeof(payload_out));
96
97 if ((sapi_buffer_read_u32_le(&payload_buf, 0U, &decoded_id) == SAPI_STATUS_OK)
98 && (decoded_id == checkpoint_id))
99 {
100 confirmed = true;
101 }
102 }
103 }
104
105 return confirmed;
106}

120static sapi_duration_ms_t remaining_budget_ms(sapi_timestamp_ms_t start_ms,
121 sapi_timestamp_ms_t now_ms,

123{

129 remaining = 0U;

131 else

139sapi_status_t sapi_channel_checkpoint(sapi_voter_t *voter, const sapi_checkpoint_config_t *config)
140{
141 sapi_status_t status = SAPI_STATUS_OK;
142 sapi_vital_message_t arrival_msg;
143 uint32_t channel_count;
144
145 channel_count = sapi_voter_get_channel_count(voter);
146
147 if ((voter == NULL) || (config == NULL))
148 {
149 status = SAPI_STATUS_INVALID_PARAM;
150 }
151 else if ((config->expected_node_count == 0U) || (config->expected_node_count > channel_count))
152 {
153 status = SAPI_STATUS_INVALID_PARAM;
154 }
155 else
156 {
157 status = build_arrival_message(config->checkpoint_id, &arrival_msg);
158 }
159
160 if (status == SAPI_STATUS_OK)
161 {
162 sapi_timestamp_ms_t start_ms = 0U;
163 uint32_t confirmed_count = 0U;
164 uint32_t i;
165 sapi_status_t last_send_status = SAPI_STATUS_OK;
166 sapi_status_t last_recv_status = SAPI_STATUS_OK;
167
168 (void)sapi_timer_now(&start_ms);
169
170 for (i = 0U; i < channel_count; i++)
171 {
172 sapi_channel_t *channel = sapi_voter_get_channel(voter, i);
173 sapi_status_t send_status = sapi_channel_send(channel, &arrival_msg, sizeof(arrival_msg));
174
175 last_send_status = send_status;
176 if (send_status == SAPI_STATUS_OK)
177 {
178 sapi_vital_message_t reply;
179 sapi_timestamp_ms_t now_ms = start_ms;
180 sapi_status_t recv_status;
181
182 (void)sapi_timer_now(&now_ms);
183 recv_status = sapi_channel_receive(
184 channel, &reply, sizeof(reply),
185 remaining_budget_ms(start_ms, now_ms, config->max_delay_ms));
186
187 last_recv_status = recv_status;
188 if ((recv_status == SAPI_STATUS_OK) && reply_confirms_checkpoint(&reply, config->checkpoint_id))
189 {
190 confirmed_count++;
191 }
192 }
193 }
194
195 if (confirmed_count >= config->expected_node_count)
196 {
197 if (config->watchdog != NULL)
198 {
199 (void)sapi_watchdog_kick(config->watchdog);
200 }
201 status = SAPI_STATUS_OK;
202 }
203 else
204 {
205 /* Diagnostic-only, fixed-size buffer (no dynamic allocation,
206 * CLAUDE.md): before this REQ-COMMON-SAFESTATE-002 permanent
207 * halt, capture the LAST channel's own send/receive outcome so
208 * a registered SAFE-level handler (see sapi_safestate.h) can
209 * actually log WHY this rendezvous failed - confirmed_count
210 * alone does not distinguish "peer never sent" (send_status
211 * failure) from "peer sent but didn't reply in time"
212 * (recv_status == SAPI_STATUS_TIMEOUT) from "replied with the
213 * wrong checkpoint_id" (recv_status == SAPI_STATUS_OK but not
214 * confirmed) - this was previously impossible to tell apart
215 * from outside this function, since no handler was ever
216 * registered for this level anywhere in this codebase and the
217 * halt itself is otherwise completely silent. */
218 char diag_message[96];
219
220 (void)snprintf(diag_message, sizeof(diag_message),
221 "checkpoint confirmed=%u expected=%u channels=%u last_send=%d last_recv=%d",
222 (unsigned int)confirmed_count, (unsigned int)config->expected_node_count,
223 (unsigned int)channel_count, (int)last_send_status, (int)last_recv_status);
224 /* Matches the pattern sapi_voter_receive() already uses on a
225 * voting disagreement: safe-state is triggered directly by
226 * the sync/vote logic itself, not left to the caller to
227 * notice and react to (REQ-CHECKPOINT-003). */
228 sapi_safestate_enter(SAPI_SAFESTATE_LEVEL_SAFE, SAPI_SAFESTATE_REASON_CHECKPOINT_TIMEOUT, __FILE__,
229 (int32_t)__LINE__, diag_message);
230 status = SAPI_STATUS_TIMEOUT;
231 }
232 }
233
234 return status;
235}
