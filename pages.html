 Cast Module - User Guide\n *\n * @section overview What is the Cast Module?\n *\n * The Cast module provides <strong>checked type conversions</strong> to prevent silent truncation\n * and overflow. Instead of C-style casts (which truncate silently), these functions\n * detect out-of-range values and return status codes.\n *\n * <strong>Key idea:</strong> Know when data is lost. Silent truncation is a security hole.\n *\n * @section quick_start Quick Start\n *\n * <tt>c\\n * \#include \\"safeapi/cast/sapi_cast.h\\"\\n *\\n * // ✗ BAD: Silent truncation\\n * int32_t large = 100000;\\n * int16_t small = (int16_t)large; // 100000 % 65536 = 34464 (wrong!)\\n *\\n * // ✓ GOOD: Checked conversion\\n * int32_t large = 100000;\\n * int16_t small;\\n * sapi_status_t rc = sapi_cast_int32_to_int16(large, \&small);\\n * if (rc == SAPI_STATUS_INVALID_PARAM) {\\n * log_error(\\"Value d too large for int16_t\\", large);\\n * return rc;\\n * }\\n * // small is guaranteed in range\\n *</tt>\n *\n * @section conversions Available Conversions\n *\n * <strong>Upcasting (safe):</strong>\n * - int8 → int16, int32, int64\n * - uint8 → uint16, uint32, uint64\n * - int16 → int32, int64\n * - uint16 → uint32, uint64\n * - etc.\n * - Always safe (no overflow possible)\n\n * <strong>Downcasting (checked):</strong>\n * - int32 → int16, int8\n * - uint32 → uint16, uint8\n * - Returns INVALID_PARAM if out of range\n * - Value only modified if success\n *\n * <strong>Signed to Unsigned:</strong>\n * - int32 → uint32 (negative → error)\n * - int8 → uint8 (negative → error)\n * - Zero and positive values OK\n *\n * <strong>Floating Point (if supported):</strong>\n * - float → int32 (checks for infinity, NaN)\n * - double → float (checks for overflow)\n * - Non-finite values return error\n *\n * @section examples Practical Examples\n *\n * ### Example 1: ADC Reading Conversion\n *\n * <tt>c\\n * uint16_t adc_raw = read_adc(); // 0-4095 (12-bit ADC)\\n *\\n * // Need to convert to temperature in 0.1°C units\\n * // Safe conversion (raw fits in uint16_t)\\n * int16_t temp_units;\\n * sapi_status_t rc = sapi_cast_uint16_to_int16(adc_raw, \&temp_units);\\n *\\n * if (rc != SAPI_STATUS_OK) {\\n * log_error(\\"ADC value out of range\\");\\n * SAPI_SAFESTATE(SAPI_SAFESTATE_LEVEL_SAFE, REASON);\\n * }\\n * // temp_units is now 0-4095 safely\\n *</tt>\n *\n * ### Example 2: CAN ID Conversion\n *\n * <tt>c\\n * uint32_t can_id = receive_can_id(); // CAN uses 29-bit or 11-bit IDs\\n *\\n * // CAN extended ID uses only 29 bits, 3 bits reserved\\n * if (can_id > 0x1FFFFFFF) { // 29-bit max\\n * log_error(\\"Invalid CAN ID: 0xx\\", can_id);\\n * return SAPI_STATUS_INVALID_PARAM;\\n * }\\n *</tt>\n *\n * ### Example 3: Pointer to Integer Conversion\n *\n * <tt>c\\n * // When storing pointers as integers (e.g., for debugging)\\n * void *ptr = get_pointer();\\n * uint32_t addr32;\\n *\\n * sapi_status_t rc = sapi_cast_ptr_to_uint32(ptr, \&addr32);\\n * if (rc != SAPI_STATUS_OK) {\\n * log_error(\\"Pointer too large for uint32_t\\");\\n * // On 64-bit system, address might not fit in 32 bits\\n * }\\n *</tt>\n *\n * @section guidelines Best Practices\n *\n * 1. <strong>Always Use Checked Casts for Downcasting</strong>\n * - Never (type)value for size reduction\n * - Always check with sapi_cast_*\n * - Handle errors, don't ignore them\n *\n * 2. <strong>Upcasting is Safe</strong>\n * - int8 → int32: safe (no loss)\n * - Can use direct cast if confident\n * - Or use cast functions for consistency\n *\n * 3. <strong>Handle Errors Appropriately</strong>\n * - For safety-critical: trigger safe-state\n * - For non-critical: log warning\n * - Always have error path\n *\n * 4. <strong>Document Assumptions</strong>\n * - Comment why cast is safe\n * - Document expected ranges\n * - Help future maintainers\n *\n * @section see_also See Also\n *\n * - @ref cast_architecture for internal design\n * - @ref types_user_guide for fixed-width types\n * - @ref status_user_guide for error handling\n *\n */

 Checksum Module - User Guide\n *\n * @section overview What is Checksum?\n *\n * The Checksum module provides <strong>data integrity verification</strong>. It computes\n * CRC32 or simple checksum to detect accidental data corruption.\n *\n * <strong>Key idea:</strong> Know when data is corrupted. Don't silently process bad data.\n *\n * @section quick_start Quick Start\n *\n * <tt>c\\n * \#include \\"safeapi/checksum/sapi_checksum.h\\"\\n *\\n * // Compute checksum\\n * uint32_t crc = sapi_checksum_crc32(data, length);\\n *\\n * // Verify checksum\\n * uint32_t stored_crc = get_stored_checksum();\\n * if (crc != stored_crc) {\\n * log_error(\\"Data corrupted: CRC mismatch\\");\\n * SAPI_SAFESTATE(SAPI_SAFESTATE_LEVEL_SAFE, REASON);\\n * }\\n *</tt>\n *\n * @section functions Functions\n *\n * <tt>c\\n * // Compute CRC32 of entire buffer\\n * uint32_t sapi_checksum_crc32(const uint8_t *data, size_t length);\\n *\\n * // Simple 8-bit checksum (XOR of all bytes)\\n * uint8_t sapi_checksum_xor8(const uint8_t *data, size_t length);\\n *\\n * // CRC16 for smaller data\\n * uint16_t sapi_checksum_crc16(const uint8_t *data, size_t length);\\n *</tt>\n *\n * @section examples Practical Examples\n *\n * ### Example 1: Message Integrity Check\n *\n * <tt>c\\n * typedef struct {\\n * uint32_t command_id;\\n * uint8_t data[256];\\n * uint32_t checksum; // Computed over all previous fields\\n * } message_t;\\n *\\n * sapi_status_t process_message(const message_t *msg) {\\n * // Compute checksum over everything except checksum field\\n * uint32_t computed = sapi_checksum_crc32((const uint8_t *)msg,\\n * offsetof(message_t, checksum));\\n *\\n * if (computed != msg->checksum) {\\n * log_error(\\"Message corrupted\\");\\n * return SAPI_STATUS_INVALID_PARAM;\\n * }\\n *\\n * execute_command(msg->command_id, msg->data);\\n * return SAPI_STATUS_OK;\\n * }\\n *</tt>\n *\n * ### Example 2: Configuration Validation\n *\n * <tt>c\\n * typedef struct {\\n * uint32_t version;\\n * char device_name[32];\\n * uint32_t timeout_ms;\\n * uint32_t crc; // Checksum of above\\n * } config_t;\\n *\\n * sapi_status_t load_config(config_t *cfg) {\\n * // Read from NVM\\n * read_nvm(0, (uint8_t *)cfg, sizeof(config_t));\\n *\\n * // Verify checksum\\n * size_t data_len = offsetof(config_t, crc);\\n * uint32_t computed = sapi_checksum_crc32((const uint8_t *)cfg, data_len);\\n *\\n * if (computed != cfg->crc) {\\n * log_error(\\"Configuration corrupted, using defaults\\");\\n * init_default_config(cfg);\\n * }\\n *\\n * return SAPI_STATUS_OK;\\n * }\\n *</tt>\n *\n * @section guidelines Best Practices\n *\n * 1. <strong>Use CRC32 for Large Data</strong>\n * - Better collision detection\n * - Slightly slower but worth it\n *\n * 2. <strong>Checksum Storage</strong>\n * - Store checksum with data\n * - Compute over all data except checksum field\n * - Verify on load/receive\n *\n * 3. <strong>When to Verify</strong>\n * - After receiving over network\n * - After reading from storage\n * - Before executing critical commands\n *\n * 4. <strong>Handling Corruption</strong>\n * - Don't try to \"fix\" corrupted data\n * - Log the corruption\n * - Either reload from backup or trigger safe-state\n *\n * @section see_also See Also\n *\n * - @ref checksum_architecture\n * - @ref nvm_user_guide for stored data integrity\n * - @ref ipc_user_guide for transmission integrity\n *\n */

 String Module - User Guide\n *\n * @section overview What is the String Module?\n *\n * The String module provides <strong>bounded string operations</strong> to prevent buffer overflows.\n * All operations specify maximum lengths and return status codes instead of silently\n * truncating or crashing.\n *\n * <strong>Key idea:</strong> Strings with known boundaries. No overflows, no truncation surprises.\n *\n * @section quick_start Quick Start\n *\n * ### 1. Include Header\n *\n * <tt>c\\n * \#include \\"safeapi/string/sapi_string.h\\"\\n *</tt>\n *\n * ### 2. Safe Copy\n *\n * <tt>c\\n * // ✗ BAD: Unbounded\\n * strcpy(dest, src); // What if src is longer than dest?\\n *\\n * // ✓ GOOD: Bounded with status code\\n * sapi_status_t rc = sapi_strncpy(dest, src, 32);\\n * if (rc != SAPI_STATUS_OK) {\\n * log_error(\\"String copy failed: s\\", sapi_status_to_string(rc));\\n * }\\n *</tt>\n *\n * ### 3. Safe Concatenation\n *\n * <tt>c\\n * char path[256];\\n * strcpy(path, \\"/home/\\");\\n *\\n * sapi_status_t rc = sapi_strncat(path, username, 256);\\n * if (rc == SAPI_STATUS_RESOURCE_EXHAUSTED) {\\n * log_error(\\"Path too long\\");\\n * return rc;\\n * }\\n *</tt>\n *\n * ### 4. Safe String Length\n *\n * <tt>c\\n * size_t len = sapi_strnlen(str, 256);\\n * if (len == 256) {\\n * log_error(\\"String not null-terminated within 256 bytes\\");\\n * }\\n *</tt>\n *\n * @section operations Operations\n *\n * <tt>c\\n * // Copy: Bounded strcpy\\n * sapi_status_t sapi_strncpy(char *dest, const char *src, size_t max_len);\\n *\\n * // Concatenate: Bounded strcat\\n * sapi_status_t sapi_strncat(char *dest, const char *src, size_t max_len);\\n *\\n * // Length: With maximum limit\\n * size_t sapi_strnlen(const char *str, size_t max_len);\\n *\\n * // Comparison: Bounded strcmp\\n * int sapi_strncmp(const char *a, const char *b, size_t max_len);\\n *\\n * // Format: Bounded sprintf\\n * sapi_status_t sapi_snprintf(char *buf, size_t max_len, const char *fmt, ...);\\n *</tt>\n *\n * @section examples Practical Examples\n *\n * ### Example 1: Command Parsing\n *\n * <tt>c\\n * void parse_command(const char *input) {\\n * char cmd[32];\\n * char arg[64];\\n *\\n * // Parse safely\\n * if (sscanf(input, \\"%31s %63s\\", cmd, arg) != 2) {\\n * log_error(\\"Invalid command format\\");\\n * return;\\n * }\\n *\\n * // Process command\\n * if (sapi_strncmp(cmd, \\"status\\", 32) == 0) {\\n * handle_status(arg);\\n * } else if (sapi_strncmp(cmd, \\"reset\\", 32) == 0) {\\n * handle_reset(arg);\\n * }\\n * }\\n *</tt>\n *\n * ### Example 2: Log Message Formatting\n *\n * <tt>c\\n * void log_event(const char *event, int code, const char *detail) {\\n * char buffer[256];\\n * sapi_status_t rc = sapi_snprintf(buffer, sizeof(buffer),\\n * \\"[u] EVENT: s (code=d) - s\\",\\n * get_timestamp(), event, code, detail);\\n *\\n * if (rc != SAPI_STATUS_OK) {\\n * // Message too long, truncated or failed\\n * log_error(\\"Log message formatting failed\\");\\n * return;\\n * }\\n *\\n * write_to_log(buffer);\\n * }\\n *</tt>\n *\n * @section guidelines Best Practices\n *\n * 1. <strong>Always Specify Maximum Length</strong>\n * - No open-ended operations\n * - Always know buffer size\n * - Use sizeof(buffer) when possible\n *\n * 2. <strong>Check Return Codes</strong>\n * - SAPI_STATUS_OK: Success, string complete\n * - SAPI_STATUS_RESOURCE_EXHAUSTED: String truncated/overflow\n * - SAPI_STATUS_INVALID_PARAM: Bad input\n *\n * 3. <strong>Use Fixed-Size Buffers</strong>\n * - Never dynamically sized strings\n * - Allocate statically\n * - Know max length at compile-time\n *\n * 4. <strong>Prefer snprintf Over sprintf</strong>\n * - snprintf has length limit\n * - sprintf doesn't (buffer overflow risk)\n * - Always use sapi_snprintf()\n *\n * @section see_also See Also\n *\n * - @ref string_architecture for internal design\n * -
