57sapi_status_t sapi_log_register_backend(const sapi_log_backend_t *backend)
58{
59 sapi_status_t lifecycle_status;
60
61 if (backend == NULL)
62 {
63 return SAPI_STATUS_INVALID_PARAM;
64 }
65 /* REQ-LIFECYCLE-001 (ADR-026): registering a backend is a setup-only
66 * action - refuse once the application's setup phase has been locked. */
67 lifecycle_status = sapi_lifecycle_check_setup_allowed();
68 if (lifecycle_status != SAPI_STATUS_OK)
69 {
70 return lifecycle_status;
71 }
72 s_backend = backend;
73 return SAPI_STATUS_OK;
74}

123void sapi_log_write_event(sapi_log_level_t level,
124 const char *site,
125 uint32_t cycle,
126 const char *source,
127 const char *destination,
128 const char *type,
129 const char *info,
130 const char *extra_fields)
131{
132 char line_storage[SAPI_LOG_EVENT_LINE_MAX_LEN];
133 char ts_storage[24];
134 sapi_string_t line;
135 sapi_string_t ts;
136 sapi_timestamp_ms_t now_ms = 0U;
137 const char *ts_cstr = NULL;
138 const char *line_cstr = NULL;
139
140 if ((s_backend == NULL) || (s_backend->write == NULL))
141 {
142 /* Same silent-no-op contract as sapi_log_write() above - avoid
143 * even the formatting work below when there is nowhere for it
144 * to go. */
145 return;
146 }
147
148 (void)sapi_string_init(&line, line_storage, sizeof(line_storage));
149 (void)sapi_string_init(&ts, ts_storage, sizeof(ts_storage));
150
151 /* Site: written directly (no leading space/key), first field. */
152 (void)sapi_string_concat(&line, "Site=");
153 (void)sapi_string_concat(&line, (site != NULL) ? site : "");
154
155 /* TIMESTAMP: best-effort - stays 0 (its declared initializer) if no
156 * sapi_timer backend is registered or the call otherwise fails; a
157 * timer problem must never prevent this event from being logged
158 * (REQ-OAL-LOG-001), so the field is degraded, not the whole call. */
159 (void)sapi_timer_now(&now_ms);
160 (void)sapi_string_from_u64(&ts, now_ms);
161 (void)sapi_string_c_str(&ts, &ts_cstr);
162 sapi_log_append_event_field(&line, "Timestamp", (ts_cstr != NULL) ? ts_cstr : "0");
163
164 sapi_log_append_event_field(&line, "Level", sapi_log_level_to_string(level));
165 sapi_log_append_event_field_u32(&line, "Cycle", cycle);
166 sapi_log_append_event_field(&line, "Source", source);
167 sapi_log_append_event_field(&line, "Destination", destination);
168 sapi_log_append_event_field(&line, "Type", type);
169 sapi_log_append_event_field(&line, "Info", info);
170
171 if (extra_fields != NULL)
172 {
173 /* extra_fields is already caller-formatted Key=Value text (see
174 * this function's own doc) - appended verbatim behind a single
175 * separating space, not wrapped in another key, so it reads as
176 * more of the same space-separated Key=Value convention rather
177 * than a nested field. */
178 (void)sapi_string_concat(&line, " ");
179 (void)sapi_string_concat(&line, extra_fields);
180 }
181
182 (void)sapi_string_c_str(&line, &line_cstr);
183 s_backend->write(level, source, (line_cstr != NULL) ? line_cstr : "");
184}
