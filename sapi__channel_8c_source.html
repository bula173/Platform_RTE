46sapi_status_t sapi_channel_compare(sapi_const_buffer_t local_result,
47 sapi_const_buffer_t peer_result,
48 sapi_channel_compare_result_t *out_result)
49{
50 sapi_status_t status = SAPI_STATUS_OK;
51
52 if (out_result == NULL)
53 {
54 status = SAPI_STATUS_INVALID_PARAM;
55 }
56 else if ((local_result.data == NULL) && (local_result.length > 0U))
57 {
58 status = SAPI_STATUS_INVALID_PARAM;
59 }
60 else if ((peer_result.data == NULL) && (peer_result.length > 0U))
61 {
62 status = SAPI_STATUS_INVALID_PARAM;
63 }
64 else if (local_result.length != peer_result.length)
65 {
66 /* REQ-COMMON-CHANNEL-002: a length difference is itself a
67 * disagreement, not an error - report it as such. */
68 *out_result = SAPI_CHANNEL_COMPARE_MISMATCH;
69 }
70 else if (local_result.length == 0U)
71 {
72 /* Both empty and same (zero) length: trivially identical.
73 * Avoided calling memcmp() with a zero count on possibly-NULL
74 * pointers, which is unspecified/undefined in the C standard even
75 * though many implementations tolerate it - defensive coding
76 * means not relying on that tolerance. */
77 *out_result = SAPI_CHANNEL_COMPARE_MATCH;
78 }
79 else if (memcmp(local_result.data, peer_result.data, local_result.length) == 0)
80 {
81 *out_result = SAPI_CHANNEL_COMPARE_MATCH;
82 }
83 else
84 {
85 *out_result = SAPI_CHANNEL_COMPARE_MISMATCH;
86 }
87
88 return status;
89}
