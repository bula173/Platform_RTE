375sapi_status_t sapi_checksum_crc64_init(sapi_crc64_polynomial_t polynomial)
376{
377 /* Verify not already initialized */
378 if (g_checksum_manager.initialized != 0U) {
379 return SAPI_STATUS_ALREADY_INITIALIZED;
380 }
381
382 /* Select appropriate lookup table */
383 switch (polynomial) {
384 case SAPI_CRC64_ERTMS:
385 g_checksum_manager.table = g_crc64_ertms_table;
386 g_checksum_manager.polynomial = SAPI_CRC64_ERTMS;
387 break;
388 case SAPI_CRC64_ISO:
389 g_checksum_manager.table = g_crc64_iso_table;
390 g_checksum_manager.polynomial = SAPI_CRC64_ISO;
391 break;
392 case SAPI_CRC64_XZ:
393 g_checksum_manager.table = g_crc64_xz_table;
394 g_checksum_manager.polynomial = SAPI_CRC64_XZ;
395 break;
396 default:
397 return SAPI_STATUS_INVALID_PARAM;
398 }
399
400 /* Initialize statistics */
401 memset(&g_checksum_manager.stats, 0, sizeof(g_checksum_manager.stats));
402 g_checksum_manager.initialized = 1U;
403
404 return SAPI_STATUS_OK;
405}

407sapi_crc64_t sapi_checksum_crc64(const uint8_t *data, size_t size)
408{
409 uint64_t crc = 0xFFFFFFFFFFFFFFFFULL; /* Initial value (all ones) */
410 size_t i;
411 uint8_t index;
412
413 /* Verify initialization */
414 if (g_checksum_manager.initialized == 0U) {
415 return 0ULL;
416 }
417
418 /* Handle NULL data pointer */
419 if (data == NULL) {
420 if (size != 0U) {
421 return 0ULL;
422 }
423 return crc;
424 }
425
426 /* Defense-in-depth against corrupted static state (e.g. a single-event
427 * upset flipping g_checksum_manager.table without also flipping
428 * .initialized) rather than a reachable API path: every branch in
429 * sapi_checksum_crc64_init() above sets .table and .initialized
430 * together, so .initialized == 1U already implies .table != NULL for
431 * any state this module's own code can produce - not coverable by a
432 * legitimate caller, and there is no public way to corrupt
433 * module-static state to hit it either. Kept rather than deleted:
434 * this guards a genuine hardware fault class relevant to SIL
435 * contexts, not merely a defensive habit. See docs/COVERAGE_REPORT.md
436 * for the accepted-exception rationale (this is the project's one
437 * documented gap against the 100% branch coverage target). */
438 if (g_checksum_manager.table == NULL) { /* GCOVR_EXCL_BR_LINE */
439 return 0ULL; /* GCOVR_EXCL_LINE - see rationale above */
440 }
441
442 /* Compute CRC using lookup table (O(1) per byte) */
443 for (i = 0U; i < size; i++) {
444 /* Extract lower 8 bits of CRC, XOR with next data byte */
445 index = (uint8_t)((crc ^ data[i]) & 0xFFU);
446
447 /* Right-shift CRC by 8 bits, XOR with table entry */
448 crc = (crc >> 8U) ^ g_checksum_manager.table[index];
449 }
450
451 /* Final XOR with all-ones */
452 crc ^= 0xFFFFFFFFFFFFFFFFULL;
453
454 /* Update statistics */
455 g_checksum_manager.stats.total_checksums++;
456
457 return crc;
458}

460sapi_status_t sapi_checksum_crc64_verify(const uint8_t *data,
461 size_t size,
462 sapi_crc64_t expected_crc,
463 sapi_checksum_result_t *result_out)
464{
465 sapi_crc64_t computed_crc;
466 sapi_status_t status;
467
468 /* Verify output buffer */
469 if (result_out == NULL) {
470 return SAPI_STATUS_INVALID_PARAM;
471 }
472
473 /* Compute CRC */
474 computed_crc = sapi_checksum_crc64(data, size);
475
476 /* Fill result structure */
477 result_out->computed = computed_crc;
478 result_out->expected = expected_crc;
479 result_out->match = (computed_crc == expected_crc) ? 1U : 0U;
480
481 /* Update statistics */
482 if (result_out->match != 0U) {
483 g_checksum_manager.stats.verification_passes++;
484 status = SAPI_STATUS_OK;
485 } else {
486 g_checksum_manager.stats.verification_failures++;
487 status = SAPI_STATUS_DATA_CORRUPTION;
488 }
489
490 return status;
491}

502sapi_status_t sapi_checksum_vital_message_create(
503 sapi_vital_message_t *msg_out,
504 uint32_t sender_id,
505 uint32_t sequence,
506 const uint8_t *payload,
507 size_t payload_size)
508{
509 sapi_timestamp_ms_t now_ms = 0U;
510
511 /* Validate inputs */
512 if (msg_out == NULL) {
513 return SAPI_STATUS_INVALID_PARAM;
514 }
515
516 if (payload_size > 248U) {
517 g_checksum_manager.stats.payload_oversize++;
518 return SAPI_STATUS_INVALID_PARAM;
519 }
520
521 if ((payload == NULL) && (payload_size != 0U)) {
522 return SAPI_STATUS_INVALID_PARAM;
523 }
524
525 /* Fill message header */
526 msg_out->sequence_number = sequence;
527 msg_out->sender_id = sender_id;
528 /* Best-effort: if no timer backend is registered, sapi_timer_now()
529 * returns SAPI_STATUS_NOT_INITIALIZED and leaves now_ms at 0 -
530 * timestamp_ms is diagnostic only (see sapi_clocksync.h's file-level
531 * note: never the basis of comparison correctness), so this is not
532 * treated as a hard failure of message creation. */
533 (void)sapi_timer_now(&now_ms);
534 msg_out->timestamp_ms = (uint32_t)(now_ms & 0xFFFFFFFFULL);
535 msg_out->payload_size = (uint8_t)payload_size;
536 msg_out->padding = 0U;
537 msg_out->reserved = 0U;
538
539 /* Clear payload area */
540 memset(msg_out->payload, 0, sizeof(msg_out->payload));
541
542 /* Copy payload */
543 if ((payload != NULL) && (payload_size > 0U)) {
544 memcpy(msg_out->payload, payload, payload_size);
545 }
546
547 /* Compute CRC-64 over entire message (including payload, excluding CRC field) */
548 msg_out->crc64 = sapi_checksum_crc64(
549 (const uint8_t *)msg_out,
550 sizeof(*msg_out) - sizeof(msg_out->crc64)
551 );
552
553 return SAPI_STATUS_OK;
554}

556sapi_status_t sapi_checksum_vital_message_verify(
557 const sapi_vital_message_t *msg,
558 uint32_t expected_sequence,
559 uint8_t *payload_out,
560 size_t payload_max_size,
561 uint8_t *payload_size_out)
562{
563 sapi_checksum_result_t check_result;
564 sapi_status_t status;
565
566 /* Validate inputs */
567 if (msg == NULL) {
568 return SAPI_STATUS_INVALID_PARAM;
569 }
570
571 if (payload_out == NULL) {
572 return SAPI_STATUS_INVALID_PARAM;
573 }
574
575 if (payload_size_out == NULL) {
576 return SAPI_STATUS_INVALID_PARAM;
577 }
578
579 /* Check CRC-64 */
580 status = sapi_checksum_crc64_verify(
581 (const uint8_t *)msg,
582 sizeof(*msg) - sizeof(msg->crc64),
583 msg->crc64,
584 &check_result
585 );
586
587 if (status != SAPI_STATUS_OK) {
588 return SAPI_STATUS_DATA_CORRUPTION;
589 }
590
591 /* Check sequence number continuity */
592 if (msg->sequence_number != expected_sequence) {
593 g_checksum_manager.stats.sequence_errors++;
594 return SAPI_STATUS_DATA_CORRUPTION;
595 }
596
597 /* Validate payload size */
598 if (msg->payload_size > payload_max_size) {
599 g_checksum_manager.stats.payload_oversize++;
600 return SAPI_STATUS_INVALID_PARAM;
601 }
602
603 /* Extract payload */
604 if (msg->payload_size > 0U) {
605 memcpy(payload_out, msg->payload, msg->payload_size);
606 }
607 *payload_size_out = msg->payload_size;
608
609 return SAPI_STATUS_OK;
610}
