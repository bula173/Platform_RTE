71static bool is_valid_handle(sapi_watchdog_t watchdog)
72{
73 const sapi_watchdog_s *slot = (const sapi_watchdog_s *)watchdog;
74
75 return (slot != NULL) && (slot >= &g_watchdog_pool[0]) && (slot < &g_watchdog_pool[SAPI_WATCHDOG_MAX_COUNT])
76 && (slot->in_use != 0U);

131sapi_status_t sapi_watchdog_create(sapi_watchdog_t *handle_out, const sapi_watchdog_config_t *config)
132{
133 size_t i;
134
135 if ((handle_out == NULL) || (config == NULL) || (config->timeout_ms == 0U) || !is_valid_action(config->action))
136 {
137 return SAPI_STATUS_INVALID_PARAM;
138 }
139 if (((config->action == SAPI_WATCHDOG_ACTION_CUSTOM) || (config->action == SAPI_WATCHDOG_ACTION_FAILOVER))
140 && (config->custom_action == NULL))
141 {
142 /* FAILOVER requires a real handler just like CUSTOM does (see
143 * sapi_watchdog_timeout_handler()'s own doc) - a FAILOVER watchdog
144 * with no handler wired up can only ever log, which is what
145 * ACTION_LOG is already for; requiring custom_action here catches
146 * that misconfiguration at create() time instead of silently
147 * degrading to a log line the first time it actually fires. */
148 return SAPI_STATUS_INVALID_PARAM;
149 }
150 if (g_manager_initialized == 0U)
151 {
152 return SAPI_STATUS_NOT_INITIALIZED;
153 }
154
155 for (i = 0U; i < (size_t)SAPI_WATCHDOG_MAX_COUNT; i++)
156 {
157 if (g_watchdog_pool[i].in_use == 0U)
158 {
159 (void)memset(&g_watchdog_pool[i], 0, sizeof(g_watchdog_pool[i]));
160 g_watchdog_pool[i].in_use = 1U;
161 g_watchdog_pool[i].config = *config;
162 *handle_out = (sapi_watchdog_t)&g_watchdog_pool[i];
163 sapi_log_write(SAPI_LOG_LEVEL_INFO, config->name, "watchdog created");
164 return SAPI_STATUS_OK;
165 }
166 }
167 return SAPI_STATUS_RESOURCE_EXHAUSTED;
168}

199sapi_status_t sapi_watchdog_kick(sapi_watchdog_t watchdog)
200{
201 sapi_watchdog_s *slot = (sapi_watchdog_s *)watchdog;
202 sapi_timestamp_ms_t now_ms = 0U;
203
204 if (!is_valid_handle(watchdog))
205 {
206 return SAPI_STATUS_INVALID_PARAM;
207 }
208 if ((slot->active == 0U) || (slot->fired != 0U))
209 {
210 /* Not running, or already fired and awaiting an explicit
211 * start() before it can be kicked again - this codebase's real
212 * status enum has no generic SAPI_STATUS_ERROR (see
213 * sapi_status.h); SAPI_STATUS_INTERNAL_ERROR is the closest
214 * "kick while not in a kickable state" signal available. */
215 return SAPI_STATUS_INTERNAL_ERROR;
216 }
217 (void)sapi_timer_now(&now_ms);
218 slot->deadline_ms = now_ms + slot->config.timeout_ms;
219 slot->last_kick_ms = now_ms;
220 slot->kicks++;
221 return SAPI_STATUS_OK;
222}

224sapi_status_t sapi_watchdog_get_status(sapi_watchdog_t watchdog, sapi_watchdog_status_t *status_out)
225{
226 const sapi_watchdog_s *slot = (const sapi_watchdog_s *)watchdog;
227 sapi_timestamp_ms_t now_ms = 0U;
228
229 if (!is_valid_handle(watchdog) || (status_out == NULL))
230 {
231 return SAPI_STATUS_INVALID_PARAM;
232 }
233 (void)sapi_timer_now(&now_ms);
234 status_out->active = slot->active;
235 status_out->kicks = slot->kicks;
236 status_out->fires = slot->fires;
237 status_out->recoveries = slot->recoveries;
238 status_out->time_since_last_kick =
239 (sapi_duration_ms_t)((now_ms >= slot->last_kick_ms) ? (now_ms - slot->last_kick_ms) : 0U);
240 status_out->time_until_fire = (sapi_duration_ms_t)((slot->deadline_ms > now_ms) ? (slot->deadline_ms - now_ms) : 0U);
241 return SAPI_STATUS_OK;
242}

261void sapi_watchdog_timeout_handler(uint32_t watchdog_id)
262{
263 sapi_watchdog_s *slot;
264
265 if (watchdog_id >= (uint32_t)SAPI_WATCHDOG_MAX_COUNT)
266 {
267 return;
268 }
269 slot = &g_watchdog_pool[watchdog_id];
270 if ((slot->in_use == 0U) || (slot->active == 0U) || (slot->fired != 0U))
271 {
272 return;
273 }
274
275 slot->fired = 1U;
276 slot->fires++;
277
278 switch (slot->config.action)
279 {
280 case SAPI_WATCHDOG_ACTION_LOG:
281 sapi_log_write(SAPI_LOG_LEVEL_ERROR, slot->config.name, "watchdog timeout");
282 break;
283 case SAPI_WATCHDOG_ACTION_SAFESTATE:
284 slot->recoveries++;
285 sapi_log_write(SAPI_LOG_LEVEL_ERROR, slot->config.name, "watchdog timeout - entering SAFE state");
286 sapi_safestate_enter(SAPI_SAFESTATE_LEVEL_SAFE, SAPI_SAFESTATE_REASON_UNSPECIFIED, __FILE__,
287 (int32_t)__LINE__, slot->config.name);
288 break; /* Not statically unreachable: sapi_safestate_enter() has no
289 * [[noreturn]]/_Noreturn attribute, so the compiler cannot
290 * prove this dead - kept for switch-statement completeness. */
291 case SAPI_WATCHDOG_ACTION_REBOOT:
292 slot->recoveries++;
293 sapi_log_write(SAPI_LOG_LEVEL_ERROR, slot->config.name, "watchdog timeout - requesting reboot");
294 sapi_safestate_enter(SAPI_SAFESTATE_LEVEL_REBOOT, SAPI_SAFESTATE_REASON_UNSPECIFIED, __FILE__,
295 (int32_t)__LINE__, slot->config.name);
296 break; /* Same non-return note as SAFESTATE above. */
297 case SAPI_WATCHDOG_ACTION_FAILOVER:
298 slot->recoveries++;
299 sapi_log_write(SAPI_LOG_LEVEL_ERROR, slot->config.name, "watchdog timeout - failover requested");
300 /* sapi_watchdog_create() requires config.custom_action != NULL
301 * for this action (see its own doc) - dispatched exactly like
302 * ACTION_CUSTOM, just under a name that documents *why* the
303 * integrator registered this watchdog (loss of a redundant
304 * peer/channel) rather than *how* it reacts, which is
305 * identical to CUSTOM's own mechanism. This used to be a dead
306 * stub ("no generic failover primitive in this framework;
307 * caller must poll sapi_watchdog_get_status()") - replaced
308 * once a real integrator (safeAPIExample's dual-channel A/B
309 * link) needed exactly this reaction and found nothing to
310 * call. */
311 if (slot->config.custom_action != NULL)
312 {
313 slot->config.custom_action(slot->config.context);
314 }
315 break;
316 case SAPI_WATCHDOG_ACTION_CUSTOM:
317 slot->recoveries++;
318 if (slot->config.custom_action != NULL)
319 {
320 slot->config.custom_action(slot->config.context);
321 }
322 break;
323 default:
324 sapi_log_write(SAPI_LOG_LEVEL_ERROR, slot->config.name, "watchdog timeout - unrecognized action");
325 break;
326 }
327}

333void sapi_watchdog_timer_tick(void)
334{
335 /* Polling design, not an ISR/hardware-timer callback: this framework's
336 * only portable time source is sapi_timer_now() (a plain "read the
337 * clock" query, not a way to register a recurring OS-level interrupt
338 * across every target this framework claims to support - POSIX,
339 * QNX, bare-metal SysTick, etc.). Giving this module its own
340 * platform-specific interrupt setup would duplicate what
341 * safeapi::timer already exists to abstract, and would break the
342 * "backends are integrator-supplied" philosophy (ADR-005) for a
343 * module whose own header was never given a backend vtable. Instead:
344 * whatever already runs periodically in the integrating application
345 * (its own sapi_timer periodic callback, or just its own main loop)
346 * is expected to call this function regularly - each call is an O(N)
347 * scan (N = SAPI_WATCHDOG_MAX_COUNT, a small fixed pool) comparing
348 * each active watchdog's deadline against the current time. */
349 sapi_timestamp_ms_t now_ms = 0U;
350 uint32_t i;
351
352 if (g_manager_initialized == 0U)
353 {
354 return;
355 }
356 (void)sapi_timer_now(&now_ms);
357 for (i = 0U; i < (uint32_t)SAPI_WATCHDOG_MAX_COUNT; i++)
358 {
359 const sapi_watchdog_s *slot = &g_watchdog_pool[i];
360
361 if ((slot->in_use != 0U) && (slot->active != 0U) && (slot->fired == 0U) && (now_ms >= slot->deadline_ms))
362 {
363 sapi_watchdog_timeout_handler(i);
364 }
365 }
366}
